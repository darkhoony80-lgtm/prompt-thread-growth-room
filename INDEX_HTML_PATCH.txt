2) updateBatchButton() 함수 전체 교체

function updateBatchButton(){
  const items=Array.isArray(window._replyItems)?window._replyItems:[];
  const eligible=items.filter(x=>!x.review_required&&!x.already_replied);
  const n=eligible.length;
  batchReply.disabled=!n||window._replyHistoryAvailable===false||window._batchRunning;
  batchReply.textContent=window._batchRunning?'일괄 답장 처리 중…':(n?`일괄 승인 후 답장 (${n})`:'일괄 승인 후 답장');
  batchNote.textContent=window._replyHistoryAvailable===false
    ?'답장 이력을 확인하지 못해 일괄 답장을 비활성화했습니다.'
    :`최근 2일 미응답 ${eligible.length}개 · API 보호를 위해 한 개씩 순차 처리합니다.`;
}
